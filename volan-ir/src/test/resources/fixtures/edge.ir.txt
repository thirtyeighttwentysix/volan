model Account -> "accounts"
  key: Uuid required [default=uuid()]
  tenant: String required [default=cuid()]
  balance: Decimal required [default=0.0]
  labels: String list [default=[]]
  token: String required [default=dbgenerated("gen_random_uuid()")]
  legacy: Int required [default=dbgenerated()]
  note: String optional [ignored]
  primary key (key, tenant) -> "account_pk"
  unique (tenant, balance) -> "account_tenant_balance"
  index (balance) btree -> "account_balance_idx"
